[solution]
void drawIsocelesTriangle() {
    for (int row = 1; row <= 7; row++) {
        int count;
        if (row <= 4) {
            count = row;
        } else {
            count = 8 - row;
        }
        for (int i = 0; i < count; i++) {
            printf("A");
        }
        printf("\n");
    }
}
