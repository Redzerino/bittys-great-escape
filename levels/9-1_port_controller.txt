[solution]
int find_max(int *arr, int size) {
    int max = *arr;
    for (int i = 1; i < size; i++) {
        arr++;
        if (*arr > max) {
            max = *arr;
        }
    }
    return max;
}

[tests]
find_max({3, 9, 4}, 3) == 9
find_max({-5, -2, -9}, 3) == -2
find_max({8}, 1) == 8
find_max({1, 5, 2, 5, 3}, 5) == 5
