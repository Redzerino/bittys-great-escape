[solution]
int findMax(int arr[], int size) {
    int max = arr[0];
    for (int i = 1; i < size; i++) {
        if (arr[i] > max) {
            max = arr[i];
        }
    }
    return max;
}

[tests]
findMax({3, 9, 4}, 3) == 9
findMax({7}, 1) == 7
findMax({1, 2, 3, 4, 5}, 5) == 5
findMax({-5, -2, -9}, 3) == -2
